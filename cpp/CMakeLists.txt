cmake_minimum_required(VERSION 3.20)
project(gafro_cpp_benchmarks LANGUAGES CXX)
set(GAFRO_CPP_DIR "" CACHE PATH "Path to the gafro-cpp repository checkout")
set(GAFRO_CPP_BUILD_DIR "" CACHE PATH "Path to its configured build tree")
if(NOT GAFRO_CPP_DIR)
    message(FATAL_ERROR "Set GAFRO_CPP_DIR to an explicit gafro-cpp checkout")
endif()
get_filename_component(GAFRO_CPP_DIR "${GAFRO_CPP_DIR}" REALPATH)
if(NOT EXISTS "${GAFRO_CPP_DIR}/src/gafro/gafro.hpp")
    message(FATAL_ERROR "GAFRO_CPP_DIR lacks src/gafro/gafro.hpp: ${GAFRO_CPP_DIR}")
endif()
get_filename_component(GAFRO_CPP_BUILD_DIR "${GAFRO_CPP_BUILD_DIR}" REALPATH)
if(NOT EXISTS "${GAFRO_CPP_BUILD_DIR}/src/gafro/gafro_package_config.hpp")
    message(FATAL_ERROR "GAFRO_CPP_BUILD_DIR lacks generated src/gafro/gafro_package_config.hpp")
endif()
find_package(Eigen3 3.4 REQUIRED)
add_executable(bench_cga_cpp bench_cga.cpp)
target_include_directories(bench_cga_cpp PRIVATE "${GAFRO_CPP_DIR}/src" "${GAFRO_CPP_BUILD_DIR}/src")
target_link_libraries(bench_cga_cpp PRIVATE Eigen3::Eigen)
target_compile_options(bench_cga_cpp PRIVATE -std=c++2c -O3 -march=native -DNDEBUG -Wno-deprecated-declarations)
target_compile_definitions(bench_cga_cpp PRIVATE
    GAFRO_BENCH_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}-cxx26"
    GAFRO_BENCH_FLAGS="-std=c++2c -O3 -march=native -DNDEBUG")
